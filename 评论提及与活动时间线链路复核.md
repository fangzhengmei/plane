class IssueActivityEndpoint(BaseAPIView):
    permission_classes = [ProjectEntityPermission]
    use_read_replica = True

    @allow_permission([ROLE.ADMIN, ROLE.MEMBER, ROLE.GUEST])
    def get(self, request, slug, project_id, issue_id):
        filters = {}
        if request.GET.get("created_at__gt", None) is not None:
            filters = {"created_at__gt": request.GET.get("created_at__gt")}

        # ========== 查询 1: Issue 活动（排除评论等）==========
        issue_activities = (
            IssueActivity.objects.filter(issue_id=issue_id)
            .filter(
                ~Q(field__in=["comment", "vote", "reaction", "draft"]),
                project__project_projectmember__member=self.request.user,
                project__project_projectmember__is_active=True,
                project__archived_at__isnull=True,
                workspace__slug=slug,
            )
            .filter(**filters)
            .select_related("actor", "workspace", "issue", "project")
        ).order_by("created_at")

        # ========== 查询 2: Issue 评论 ==========
        issue_comments = (
            IssueComment.objects.filter(issue_id=issue_id)
            .filter(
                project__project_projectmember__member=self.request.user,
                project__project_projectmember__is_active=True,
                project__archived_at__isnull=True,
                workspace__slug=slug,
            )
            .filter(**filters)
            .order_by("created_at")
            .select_related("actor", "issue", "project", "workspace")
            .prefetch_related(
                Prefetch("comment_reactions", queryset=CommentReaction.objects.select_related("actor"))
            )
        )

        # ========== 分支 1: 只返回属性活动 ==========
        if request.GET.get("activity_type", None) == "issue-property":
            issue_activities = issue_activities.prefetch_related(
                Prefetch("issue__issue_intake", queryset=IntakeIssue.objects.only(...), to_attr="source_data")
            )
            issue_activities = IssueActivitySerializer(issue_activities, many=True).data
            return Response(issue_activities, status=status.HTTP_200_OK)

        # ========== 分支 2: 只返回评论 ==========
        if request.GET.get("activity_type", None) == "issue-comment":
            issue_comments = IssueCommentSerializer(issue_comments, many=True).data
            return Response(issue_comments, status=status.HTTP_200_OK)

        # ========== 分支 3: 合并返回（默认）==========
        result_list = sorted(
            chain(issue_activities, issue_comments),
            key=lambda instance: instance["created_at"],
        )
        return Response(result_list, status=status.HTTP_200_OK)

protected buildActivityAndCommentItems(issueId: string): TIssueActivityComment[] | undefined {
  const activityComments: TIssueActivityComment[] = [];
  const currentStore = this.serviceType === EIssueServiceType.EPICS 
    ? this.store.issue.epicDetail 
    : this.store.issue.issueDetail;

  const activities = this.getActivitiesByIssueId(issueId);
  const comments = currentStore.comment.getCommentsByIssueId(issueId);  // ✅ 从评论 Store 取

  // ========== 活动映射 ==========
  activities.forEach((activityId) => {
    const activity = this.getActivityById(activityId);
    const type =
      activity.field === "state" ? EActivityFilterType.STATE :
      activity.field === "assignees" ? EActivityFilterType.ASSIGNEE :
      activity.field === null ? EActivityFilterType.DEFAULT :
      EActivityFilterType.ACTIVITY;
    activityComments.push({
      id: activity.id,
      activity_type: type,  // ✅ 添加 activity_type 字段
      created_at: activity.created_at,
    });
  });

  // ========== 评论映射 ==========
  comments.forEach((commentId) => {
    const comment = currentStore.comment.getCommentById(commentId);
    activityComments.push({
      id: comment.id,
      activity_type: EActivityFilterType.COMMENT,  // ✅ 标记为评论类型
      created_at: comment.created_at,
    });
  });

  return activityComments;
}

protected sortActivityComments(items, sortOrder) {
  return orderBy(items, (e) => new Date(e.created_at || 0), sortOrder);  // ✅ 前端排序
}

getActivityAndCommentsByIssueId = computedFn((issueId, sortOrder) => {
  const baseItems = this.buildActivityAndCommentItems(issueId);
  if (!baseItems) return undefined;
  return this.sortActivityComments(baseItems, sortOrder);

export const filterActivityOnSelectedFilters = (
  activity: TIssueActivityComment[],
  filters: TActivityFilters[]
): TIssueActivityComment[] =>
  activity.filter((activity) => {
    if (activity.activity_type === EActivityFilterType.DEFAULT) return true;
    return filters.includes(activity.activity_type as TActivityFilters);

### 4.5 对 Mention 过滤能力的影响

#### ✅ 支持的过滤能力
- **按 Description 提及过滤 Issue**：`IssueMention.objects.filter(mention_id=user_id).values_list("issue_id", flat=True)`
- **按 Issue 查询被提及用户**：`IssueMention.objects.filter(issue_id=issue_id).values_list("mention_id", flat=True)`
- **按项目 + 用户过滤**：`IssueMention.objects.filter(project_id=project_id, mention_id=user_id)`

#### ❌ 不支持的过滤能力
- **按 Comment 提及过滤 Issue**：无法查询"在评论中提及过我的 Issue"
- **按 Comment 查询被提及用户**：需要解析每条评论的 HTML 才能获取
- **全局提及搜索**：无法在工作区级别搜索"某人被提及的所有 Issue（含评论提及）"
- **提及统计**：无法统计某个用户在评论中被提及的次数

### 4.6 过滤能力缺失的业务影响

| 场景 | 当前能力 | 影响 |
|-----|---------|------|
| 用户查看"所有@我的 Issue" | 只返回 Description 中@的 | 遗漏了评论中@的，用户可能错过重要提及 |
| 项目管理员查看"某人参与的 Issue" | 只包含 Description 提及 | 统计不准确 |
| 通知已读未读状态同步 | 不影响 | 通知是单独的 Notification 表 |
| 提及用户自动订阅 | ✅ 支持（两类提及都支持） | 评论提及也会添加订阅者（第 467-469 行） |

### 4.7 架构决策分析

**为什么 Description 提及和 Comment 提及采用不同的存储策略？**

1. **历史演进**：Description 提及先实现，Comment 提及后实现，TODO 注释表明计划后续补充
2. **数据量考虑**：评论提及数量远大于 Description 提及，可能担心数据膨胀
3. **查询需求优先级**：按提及过滤 Issue 的需求优先级较低，先通过 HTML 解析满足通知需求
4. **性能权衡**：每条评论创建都解析 HTML 并写入数据库 vs 按需解析（通知时解析一次）
